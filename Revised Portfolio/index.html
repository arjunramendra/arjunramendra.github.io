@RequestMapping("/api/v1")
@RequiredArgsConstructor

  @GetMapping("/health")
  public ResponseEntity<Status> health() {
    return ResponseEntity
      .ok(orchestrator
        .checkAll()
        .withLatency(42ms));
  }

  @PostMapping("/process")
  @RateLimited(rps = 10_000)
  public Flux<Result> process(
    @RequestBody Payload payload) {
    return orchestrator
      .processAsync(payload);
  }
}
